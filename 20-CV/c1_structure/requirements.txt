# CV 提取 MVP — 独立部署依赖清单
# 实测通过：Python 3.10.9 / Windows / miniforge(paddle 环境)
#
# 安装： python -m pip install -r requirements.txt
# 自检： python check_env.py
#
# 说明：下面是「直接 import 用到的」包；传递依赖（pydantic / starlette / anyio /
#       pypdfium2 / PyYAML / requests / prettytable 等）由 pip 自动解析，不必手列。
#       GPU 服务器请把 paddlepaddle 换成对应 CUDA 版本的 paddlepaddle-gpu。

# ---- Web 服务 ----
fastapi==0.139.0
uvicorn==0.40.0
jinja2==3.1.6                     # 对照报告 HTML 模板渲染（extract_report/templates/，autoescape）
# 上传走 base64 JSON（await request.json()），因此不需要 python-multipart；
# StaticFiles / FileResponse 在 Starlette 1.x 用 anyio，也不需要 aiofiles。

# ---- QoderCN Agent SDK（AI 规则生成；国际版可选装 qoder-agent-sdk）----
qodercn-agent-sdk==1.0.17

# ---- Paddle 推理（核心，体积最大）----
# paddleocr 3.7.0 只声明 paddlex[ocr-core]（仅够 OCR 文字识别），并不会装 paddlepaddle。
# 图片/扫描件路线的 PP-StructureV3 门禁要完整 ocr extra（21 包，含公式/表格后处理依赖），
# 故显式装 paddlex[ocr]；ocr-core 是其子集，opencv-contrib 等锁版本不变。
paddlepaddle==3.2.0
paddleocr==3.7.0
paddlex[ocr]==3.7.2               # 约束：>=3.7.0,<3.8.0，与 paddleocr 必须同代

# ---- 图像处理 ----
# 注意：cv2 来自 opencv-contrib-python（paddlex 的 ocr extra 指定），
#       不是 opencv-python；两个都装会造成 cv2 冲突。
opencv-contrib-python==4.10.0.84
numpy==2.2.6
Pillow==12.3.0

# ---- PDF / 文本表格 ----
pdfplumber==0.11.10               # 文本层表格 + 单元格 bbox（本项目直接用，paddle 不带）
PyMuPDF==1.28.2                   # pip 名 PyMuPDF，import 名 pymupdf；PDF 首页渲染

# ---- 平台数据库（MySQL，数据链路设计见 docs/design-data-pipeline.md）----
PyMySQL==1.2.4                    # 纯 Python MySQL 驱动；store/ 与 loader 共用驱动接缝的首个实现
# MySQL 8 默认认证 caching_sha2_password：非 SSL 首次认证走 RSA 密钥交换，
# PyMySQL 此路径必须有 cryptography（同 paddlepaddle 例：不直接 import 也须显式列出）。
cryptography==50.0.2              # 环境已随 paddlex 装入，此处锁版本防漂移
